```python
class AlwaysTopWindow(QWidget):
    """置顶窗口基类
    特性：
    - QWindowStaysOnTopHint — 始终保持最前
    - 无边框 (FramelessWindowHint) — 极简
    - 可拖拽（mousePressEvent + mouseMoveEvent）
    - 可选半透明 (WA_TranslucentBackground)
    - QTimer 自动刷新（5 秒间隔）
    """
    
    def __init__(self):
        self.setWindowFlags(
            Qt.WindowStaysOnTopHint | 
            Qt.FramelessWindowHint | 
            Qt.Tool
        )
        self.timer = QTimer()
        self.timer.timeout.connect(self.refresh)
        self.timer.start(5000)
```

### 5.7 市场操作引擎（GClass68 — 2447 行）

#### 阵营税费矩阵

不同阵营的设施提供不同的制造/反应/发明时间减免：

```
加达里设施 → 减免受 "加达里星舰工程学" 影响
盖伦特设施 → 减免受 "盖伦特星舰工程学" 影响
米玛塔尔设施 → 减免受 "米玛塔尔星舰工程学" 影响
艾玛设施 → 减免受 "艾玛星舰工程学" 影响
```

```csharp
// GClass68.smethod_6 — 获取设施费率配置
// 输入: charName, facilityName
// 输出: GClass70 { 各阵营技能等级, 费率 }

// 费率公式:
// 制造时间减免 = 1.0 - baseRate × (1 - level × 0.02) × systemIndex
//   baseRate: 0.85(普通) / 0.80(好) / 0.70(最佳)
// 反应时间减免 = 1.0 - baseRate × (1 - level × 0.02) × systemIndex
//   baseRate: 0.97(普通) / 0.96(好) / 0.95(最佳)
```

```python
def calc_trade_score(
    type_id,

    sell_price_type="sell",
    char_config=None,
    quantity=1,
):
    """贸易评分计算
    
    1. 查买价（buy_hub）
    2. 查卖价（sell_hub）  
    3. 查体积
    4. 计算费率（buy/sell 分开计算声望）
    5. 买入成本 = 买价 × 数量 × (1 + broker_fee%)
    6. 卖出收入 = 卖价 × 数量 × (1 - broker_fee% - relist% - tax%)
    7. 毛利 = 卖出收入 - 买入成本
    8. 评分 = min(利润率×5, 50) + min(成交量/500万, 1)×50
```
